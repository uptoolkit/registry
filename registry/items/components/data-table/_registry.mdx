Composable TanStack Table primitives for CRUD admin UIs.

```tsx
import { DataTable } from "@/components/ui/data-table";
import { DataTableToolbar } from "@/components/ui/data-table-toolbar";
import { DataTablePagination } from "@/components/ui/data-table-pagination";
import { useCrudDataTable } from "@/hooks/use-crud-data-table";

const { table } = useCrudDataTable({ columns, data });

<>
  <DataTableToolbar table={table} />
  <DataTable table={table} />
  <DataTablePagination table={table} />
</>;
```

`DataTableToolbar` includes global search, per-column filters, and a column
visibility menu. `DataTableColumnHeader` adds sort and hide controls to each
sortable column.

Install `@tanstack/react-table` in your app:

```bash
npm install @tanstack/react-table
```
